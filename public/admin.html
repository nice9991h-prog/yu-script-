} app.use(helmet({ contentSecurityPolicy: false })); app.use(compression()); app.use(express.json({ limit: '100kb' })); app.use(cookieParser()); const authLimit = rateLimit({ windowMs: 15 * 60 * 1000, max: 20, standardHeaders: true, legacyHeaders: false, message: { error: 'Too many requests. Please try again later.' } }); const token = (id) => jwt.sign({ adminId: id }, jwtSecret, { expiresIn: '2h' }); function auth(req, res, next) { try { const cookie = req.cookies && req.cookies.yu_admin; if (!cookie) return res.status(401).json({ error: 'Authentication required' }); req.admin = jwt.verify(cookie, jwtSecret); return next(); } catch (error) { return res.status(401).json({ error: 'Authentication required' }); } } function settingMap() { return Object.fromEntries( db.prepare('SELECT key, value FROM site_settings').all().map((x) => [x.key, x.value]) ); } function publicProduct(product) { return { ...product, price: Number(product.price) / 100, active: !!product.active }; } function generateKeyValue() { return 'YU-' + [1, 2, 3].map(() => crypto.randomBytes(2).toString('hex').toUpperCase()).join('-'); } function issueKey(orderId, productId) { const keyValue = generateKeyValue(); const hash = crypto.createHash('sha256').update(keyValue).digest('hex'); const existing = db.prepare('SELECT id FROM purchase_keys WHERE key_hash=?').get(hash); if (existing) return issueKey(orderId, productId); db.prepare('INSERT INTO purchase_keys(id, order_id, product_id, key_hash, status) VALUES(?,?,?,?,?)').run( crypto.randomUUID(), orderId, productId, hash, 'unused' ); return keyValue; } app.get('/api/site', (req, res) => { const settings = settingMap(); const products = db.prepare('SELECT * FROM products WHERE active=1 ORDER BY id DESC').all().map(publicProduct); res.json({ settings, products }); }); app.post('/api/orders', (req, res) => { const productId = Number(req.body.productId); const email = String(req.body.email || '').trim(); const product = db.prepare('SELECT * FROM products WHERE id=? AND status=?').run('used', keyRecord.id, 'unused'); if (affected.changes !== 1) { return res.status(400).json({ error: 'Key has already been used' }); } return res.json({ productName: keyRecord.product_name, downloadUrl: keyRecord.download_url || '#' }); }); app.post('/api/admin/login', authLimit, (req, res) => { const email = String(req.body.email || '').trim(); const password = String(req.body.password || ''); const admin = db.prepare('SELECT * FROM admins WHERE email=?').get(email); if (!admin || !bcrypt.compareSync(password, admin.password_hash)) { return res.status(401).json({ error: 'Invalid credentials' }); } res.cookie('yu_admin', token(admin.id), { httpOnly: true, sameSite: 'strict', secure: process.env.NODE_ENV === 'production', maxAge: 2 * 60 * 60 * 1000 }); return res.json({ ok: true }); }); app.post('/api/admin/logout', (req, res) => { res.clearCookie('yu_admin'); res.json({ ok: true }); }); app.get('/api/admin/dashboard', auth, (req, res) => { const q = (sql) => db.prepare(sql).get(); const products = db.prepare('SELECT * FROM products ORDER BY id DESC').all(); const orders = db.prepare(` SELECT o.*, p.name, p.price FROM orders o JOIN products p ON p.id = o.product_id ORDER BY o.created_at DESC `).all(); const productCount = q('SELECT COUNT(*) AS n FROM products').n; const stockTotal = q('SELECT COALESCE(SUM(stock), 0) AS n FROM products').n; const ordersCount = q('SELECT COUNT(*) AS n FROM orders').n; const paidOrders = q("SELECT COUNT(*) AS n FROM orders WHERE status='Paid'").n; const availableKeys = q("SELECT COUNT(*) AS n FROM purchase_keys WHERE status='unused'").n; const usedKeys = q("SELECT COUNT(*) AS n FROM purchase_keys WHERE status='used'").n; res.json({ products: Number(productCount), stock: Number(stockTotal), ordersCount: Number(ordersCount), paidOrders: Number(paidOrders), availableKeys: Number(availableKeys), usedKeys: Number(usedKeys), orders, productRows: products, settings: settingMap() }); }); app.post('/api/admin/products', auth, (req, res) => { const body = req.body || {}; const name = String(body.name || '').trim(); const description = String(body.description || '').trim(); const category = String(body.category || '').trim(); const price = Number(body.price || 0); const stock = Math.max(0, Number(body.stock || 0)); if (!name || !description || !category) { return res.status(400).json({ error: 'Name, description and category are required' }); } const result = db.prepare('INSERT INTO products(name, description, category, price, stock, image_url, download_url) VALUES(?,?,?,?,?,?,?)').run( name, description, category, Math.round(price * 100), stock, body.imageUrl || '', body.downloadUrl || '' ); return res.status(201).json({ id: result.lastInsertRowid }); }); app.put('/api/admin/products/:id', auth, (req, res) => { const body = req.body || {}; const id = Number(req.params.id); db.prepare('UPDATE products SET name=?, description=?, category=?, price=?, stock=?, image_url=?, download_url=?, active=?
